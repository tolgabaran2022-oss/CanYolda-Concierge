Create a single centralized image resolver if necessary.

Example logic:

function resolveAnimalImage(item) {
  const value =
    item.image_url ??
    item.imageUrl ??
    item.photo_url ??
    item.photoUrl ??
    item.animal_image ??
    item.animalImage ??
    item.images?.[0] ??
    item.photos?.[0] ??
    null;

  if (typeof value === 'string' && value.trim().length > 0) {
    return value.trim();
  }

  if (value?.url && typeof value.url === 'string') {
    return value.url;
  }

  return null;
}

IMPORTANT:
This compatibility resolver is for reading existing records.

All NEW status records must use the canonical database image field.
